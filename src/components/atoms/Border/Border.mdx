# Borders & Radius

Define a estética de arredondamento e contorno dos componentes.

export const BorderRadius = () => {
  const colors = getThemeVariables('color')
  const borderRadius = getThemeVariables('radius')

  return (
    <div className="grid grid-cols-2 lg:grid-cols-4 gap-lg">
      {Object.entries(borderRadius).map(([key, value]) => (
        <div key={key} className="flex flex-col items-center p-md bg-neutral-900 border border-neutral-800 rounded-sm">
          <div
            className="flex justify-center items-center shadow-md mb-md transform hover:scale-105 transition-transform"
            style={{
              borderRadius: value,
              backgroundColor: colors.primary || '#d85f5f',
              width: '120px',
              height: '120px'
            }}
          />
          <span className="text-content-fg font-bold">--radius-{key}</span>
          <span className="text-content-fg-subcontent text-sm">{value}</span>
        </div>
      ))}
    </div>
  )
}
